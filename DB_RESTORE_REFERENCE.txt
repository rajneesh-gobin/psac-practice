-- Trigger to keep reply_count in sync:
CREATE OR REPLACE FUNCTION public.update_reply_count()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  IF (TG_OP = 'INSERT') THEN
    UPDATE public.forum_posts SET reply_count = reply_count + 1 WHERE id = NEW.post_id;
  ELSIF (TG_OP = 'DELETE') THEN
    UPDATE public.forum_posts SET reply_count = GREATEST(0, reply_count - 1) WHERE id = OLD.post_id;
  END IF;
  RETURN NULL;
END;
$$;

DROP TRIGGER IF EXISTS trg_reply_count ON public.forum_replies;

-- ── verify_student_pin ───────────────────────────────────────────────────
-- Called by: engine/auth.js when student logs in via username + 4-digit PIN.
-- Returns: JSON object { ok, locked, secsLeft, attemptsLeft, error, student }
-- Must be created as SECURITY DEFINER so it can read students.pin.

CREATE OR REPLACE FUNCTION public.verify_student_pin(
  p_username TEXT,
  p_pin      TEXT
)
RETURNS JSONB
LANGUAGE plpgsql SECURITY DEFINER AS $$
DECLARE
  v_student   public.students%ROWTYPE;
  v_max_tries CONSTANT INT := 5;
  v_lockout   CONSTANT INT := 300; -- 5 minutes in seconds
  v_attempts  INT := 0;
  -- NOTE: implement attempt tracking via a separate table or within students.settings if needed
BEGIN
  SELECT * INTO v_student
  FROM public.students
  WHERE lower(username) = lower(p_username)
  LIMIT 1;

  IF NOT FOUND THEN
    RETURN jsonb_build_object('ok', false, 'error', 'User not found');
  END IF;

  -- Simple PIN check (plaintext for dev; replace with crypt() for prod):
  IF v_student.pin = p_pin OR crypt(p_pin, v_student.pin) = v_student.pin THEN
    RETURN jsonb_build_object('ok', true, 'student', row_to_json(v_student)::jsonb);
  ELSE
    RETURN jsonb_build_object('ok', false, 'error', 'Incorrect PIN', 'attemptsLeft', v_max_tries);
  END IF;
END;
$$;

-- ── set_student_pin ──────────────────────────────────────────────────────
-- Called by: engine/store.js when parent sets/changes a student PIN.
-- Hashes PIN with bcrypt before storing.

CREATE OR REPLACE FUNCTION public.set_student_pin(
  p_student_id UUID,
  p_pin        TEXT
)
RETURNS VOID
LANGUAGE plpgsql SECURITY DEFINER AS $$
BEGIN
  UPDATE public.students
  SET pin = crypt(p_pin, gen_salt('bf'))
  WHERE id = p_student_id;
END;
$$;

-- Enable pgcrypto extension (required for crypt/gen_salt):
CREATE EXTENSION IF NOT EXISTS pgcrypto;

SECTION 6 — APP ARCHITECTURE NOTES (for AI assistants)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

GENERAL
  - Pure vanilla JS + Tailwind CSS. No React, Vue, bundler, or build step.
  - Single HTML file (index.html) loads all scripts in <head>/<body> order.
  - All JS is globally scoped IIFEs (const Foo = (() => { ... return {...}; })())
  - Dark mode via Tailwind class="dark" on <html>

SUBJECTS & QUESTIONS
  - Global array: STATIC_QUESTIONS — all subject files push into it.
  - Each question: { id, chapterId, subjectId, grade, type, difficulty, ... }
  - Types: MCQ, TF (true/false), NUM (numeric), MATCH (matching pairs)
  - Difficulty 1–4; Level 4 MUST be a narrative word problem.
  - SUBJECT_PACKS global — array of subject pack descriptors (id, name, grade, chapters[])
  - Question file naming: subjects/grade{N}-{subject}/questions/ch{NN}_{name}.js
  - Each subject folder has _manifest.js that pushes pack metadata into SUBJECT_PACKS

STUDENT AUTH FLOW
  - Students do NOT use Supabase Auth.
  - Login: username + 4-digit PIN → verify_student_pin() RPC
  - Session stored in localStorage under key: mm_student_session
  - Session has: { studentId, name, grade, sessionVersion, expiresAt }
  - On resume: re-fetches students.session_version; mismatch = force logout

PARENT/TEACHER AUTH FLOW
  - Full Supabase Auth (email+password or Google OAuth)
  - Profile fetched from profiles table on sign-in
  - Family fetched from families table (one per parent)

CALENDAR / TIMETABLE
  - engine/calendar.js — Calendar IIFE
  - study_schedules: one row per student (settings: { weeks, studyDays, subjectHours, mixed })
  - schedule_entries: one row per day per session
  - entry_type: 'study' (auto-generated) | 'exam' | 'holiday' | 'blocked' | 'other'
  - localStorage cache key: mm_schedule_{studentId} (fallback if Supabase slow)
  - Single-subject mode: weighted rotation, exactly 1 subject per study day
  - Mixed mode: multiple subjects per day, proportional to hours setting

FORUM
  - engine/forum.js — Forum IIFE
  - Categories: general, maths, english, science, french, history, tips, suggest, report
  - reply_count maintained by DB trigger (trg_reply_count)
  - Likes: localStorage only (mm_forum_likes key), not stored in DB
  - Delete: author-name match (no secure auth check — by design for simplicity)
  - Cooldown: 30 seconds between posts; 30 seconds between replies

PROGRESS
  - engine/store.js — Store IIFE
  - All progress in student_progress.data (single JSONB blob per student)
  - Saved on a debounced timer (background upsert)
  - chapters key: { [chapterId]: { attempts, correct, lastDate, difficulty stats } }

ADMIN
  - engine/admin.js — AdminPanel IIFE
  - Accessible only when profiles.role = 'admin' or 'super_admin'
  - Controls: disable grades/subjects/chapters, manage members, plans, payments
